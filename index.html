// Reproducción de voz en ESPAÑOL DE MÉXICO (es-MX) function speakEs(text) { if ('speechSynthesis' in window) { const utterance = new SpeechSynthesisUtterance(text); utterance.lang = 'es-MX'; // Configurado específicamente para México // Buscar una voz nativa de México si está disponible en el navegador const voices = window.speechSynthesis.getVoices(); const mxVoice = voices.find(v => v.lang === 'es-MX' || v.lang.includes('MX')); if (mxVoice) { utterance.voice = mxVoice; } window.speechSynthesis.cancel(); window.speechSynthesis.speak(utterance); } else { alert('Tu navegador no soporta la reproducción de audio.'); } }
